/*
 * Instalación/verificación manual.
 * Ejecutar una vez después de pegar el archivo para comprobar que las
 * cuatro hojas existen. NO borra ni modifica registros.
 */
function instalarEndpointsMovimientosTaller() {
  var ss = SpreadsheetApp.openById(DM_TALLER_MOV_DB_ID_);
  var requeridas = [
    "MOVIMIENTOS_SUBIDAS",
    "MOVIMIENTOS_BAJADAS",
    "MOVIMIENTOS_CAMBIO_EQUIPO",
    "MOVIMIENTOS_MOVILIZACIONES"
  ];
  var faltantes = requeridas.filter(function(nombre){ return !ss.getSheetByName(nombre); });
  if (faltantes.length) throw new Error("Faltan hojas de Taller: " + faltantes.join(", "));
  return "OK — endpoints de Taller habilitados. Las 4 hojas existen y quedan disponibles para la app.";
}

/*
 * Diagnóstico opcional. Devuelve cantidad de registros por hoja.
 */
function diagnosticarEndpointsMovimientosTaller() {
  var ss = SpreadsheetApp.openById(DM_TALLER_MOV_DB_ID_);
  var nombres = [
    "MOVIMIENTOS_SUBIDAS",
    "MOVIMIENTOS_BAJADAS",
    "MOVIMIENTOS_CAMBIO_EQUIPO",
    "MOVIMIENTOS_MOVILIZACIONES"
  ];
  var out = {};
  nombres.forEach(function(nombre){
    var sh = ss.getSheetByName(nombre);
    out[nombre] = sh ? Math.max(0, sh.getLastRow() - 1) : "NO EXISTE";
  });
  return JSON.stringify(out, null, 2);
}
